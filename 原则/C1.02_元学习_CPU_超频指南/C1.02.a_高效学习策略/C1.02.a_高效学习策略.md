let rows = [];

if (folderObj && folderObj.children) {
    // 遍历当前文件夹下的所有子项并按名称排序
    const children = folderObj.children.sort((a, b) => a.name.localeCompare(b.name));

    for (let child of children) {
        // 1. 判断是否为文件夹 (child.children)
        // 2. 判断文件夹名称是否在排除列表 (excludeList) 中
        if (child.children && !excludeList.includes(child.name)) {
            
            const subFolderName = child.name;
            const subFolderPath = child.path;
            
            // 构造预期的同名文件路径：子文件夹路径/子文件夹名.md
            const expectedFilePath = `${subFolderPath}/${subFolderName}.md`;
            
            // 尝试通过 Dataview 获取该文件的元数据
            const page = dv.page(expectedFilePath);
            
            let desc = "";
            let nameDisplay = subFolderName; // 默认只显示文字名称

            if (page) {
                // 情况 1: 同名文件存在
                desc = page.description ? page.description : "";
                
                // 将第一列设为可点击的链接
                nameDisplay = page.file.link;
            } else {
                // 情况 2: 同名文件不存在
                desc = ""; 
            }

            // 添加一行数据
            rows.push([nameDisplay, desc]);
        }
    }
